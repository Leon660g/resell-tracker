<script>
let inventory = [];
function renderItems() {
    const list = document.getElementById('inventoryList');
    if (inventory.length === 0) { list.innerHTML = '<p class="empty">No items yet — add your first one above!</p>'; return; }
    list.innerHTML = inventory.map((item, index) => `<div class="item"><div><strong>${item.name}</strong> — £${item.price.toFixed(2)} <button class="btn-status" onclick="toggleStatus(${index})">${item.status}</button></div><button class="btn-delete" onclick="deleteItem(${index})">🗑️ Delete</button></div>`).join('');
}
function addItem() {
    const name = document.getElementById('itemName').value.trim();
    const price = parseFloat(document.getElementById('buyPrice').value);
    const status = document.getElementById('itemStatus').value;
    if (!name || isNaN(price)) return alert('Fill in all fields!');
    inventory.push({ name, price, status });
    saveToDB(); renderItems();
    document.getElementById('itemName').value = ''; document.getElementById('buyPrice').value = '';
}
function toggleStatus(index) {
    const statuses = ['available', 'pending', 'sold'];
    const current = statuses.indexOf(inventory[index].status);
    inventory[index].status = statuses[(current + 1) % 3];
    saveToDB(); renderItems();
}
function deleteItem(index) {
    if (confirm('Delete this item?')) { inventory.splice(index, 1); saveToDB(); renderItems(); }
}
function saveToDB() {
    fetch('/api/user/save', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ dataType: 'inventory', data: { inventory } }) });
}
async function loadFromDB() {
    const res = await fetch('/api/user/load/inventory');
    const result = await res.json();
    if (result.data && result.data.inventory) { inventory = result.data.inventory; renderItems(); }
}
function logout() { window.location.href = '/api/logout'; }
window.addEventListener('load', loadFromDB);
</script>

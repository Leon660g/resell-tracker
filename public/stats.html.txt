<script>
async function loadStats() {
    const invRes = await fetch('/api/user/load/inventory');
    const salesRes = await fetch('/api/user/load/sales');
    const invData = await invRes.json();
    const salesData = await salesRes.json();
    const items = invData.data?.inventory?.length || 0;
    const sales = salesData.data?.sales || [];
    document.getElementById('totalItems').textContent = items;
    document.getElementById('totalSales').textContent = sales.length;
    const totalProfit = sales.reduce((sum, s) => sum + s.profit, 0);
    document.getElementById('totalProfit').textContent = totalProfit.toFixed(2);
}
function logout() { window.location.href = '/api/logout'; }
window.addEventListener('load', loadStats);

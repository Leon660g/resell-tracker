<script>
fetch('/api/me').then(r=>r.json()).then(res=>{if(res.isAdmin){document.getElementById('accessDenied').classList.add('hidden');document.getElementById('adminContent').classList.remove('hidden');loadAllData();}});
function loadAllData(){fetch('/api/admin/alldata').then(r=>r.json()).then(res=>{
    const notifContainer=document.getElementById('notificationsList');
    const unreadCount=res.notifications.filter(n=>!n.read).length;
    const badge=document.getElementById('notifBadge');
    if(unreadCount>0){badge.style.display='inline';badge.textContent=unreadCount+' New';}
    if(res.notifications.length===0){notifContainer.innerHTML='<p style="color:#94a3b8;">No notifications yet.</p>';}
    else{notifContainer.innerHTML=res.notifications.map(n=>`<div class="notification ${!n.read?'unread':''}"><strong>🚨 ${n.message}</strong><div class="notification-time">${n.timestamp}</div></div>`).join('');}
    res.users.forEach(u=>{document.getElementById('usersBody').innerHTML+=`<tr><td>${u.id}</td><td>${u.username}</td><td>${u.created_at}</td></tr>`;});
    res.allUserData.forEach(d=>{const preview=d.data_json.length>60?d.data_json.slice(0,60)+'...':d.data_json;document.getElementById('dataBody').innerHTML+=`<tr><td>${d.username}</td><td>${d.data_type}</td><td>${preview}</td><td>${d.updated_at}</td></tr>`;});
    res.masterRecords.forEach(d=>{document.getElementById('masterBody').innerHTML+=`<tr><td>${d.category}</td><td>${d.label}</td><td>${d.content}</td><td>${d.created_at}</td></tr>`;});
});}
function markAllRead(){fetch('/api/admin/notifications/read',{method:'POST'}).then(()=>loadAllData());}
function addRecord(){fetch('/api/admin/master/add',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({category:document.getElementById('cat').value,label:document.getElementById('label').value,content:document.getElementById('content').value,notes:document.getElementById('notes').value})}).then(()=>{alert('✅ Saved!');location.reload();});}
function logout(){window.location.href='/api/logout';}
